系統設定

正式使用方式：先在 Google 試算表建立工作表，再把 Apps Script 部署成 Web App，將網址貼在這裡。之後每次病患領取，系統會先本機存檔，再自動嘗試同步。
function doPost(e) { const sheet = SpreadsheetApp.openById('你的 Google Sheet ID').getSheetByName('Records'); const data = JSON.parse(e.postData.contents); if (data.action === 'test') { return ContentService.createTextOutput(JSON.stringify({ ok: true, message: 'test ok' })) .setMimeType(ContentService.MimeType.JSON); } if (data.action === 'append_record') { sheet.appendRow([ new Date(), data.clinicName || '', data.record.patientName || '', data.record.patientCode || '', data.record.patientPhone || '', data.record.date || '', data.record.qty || '', data.record.signer || '', data.record.remark || '', data.record.signature || '' ]); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ ok: false, message: 'unknown action' })) .setMimeType(ContentService.MimeType.JSON); }
